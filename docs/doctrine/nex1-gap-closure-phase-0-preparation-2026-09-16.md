```bash
openssl genpkey -algorithm ed25519 -outform DER | xxd -p -c 256 > founder.key.hex
openssl pkey -in <(xxd -r -p < founder.key.hex) -pubout -outform DER | xxd -p -c 256 > founder.pub.hex

```

Option B · Node.js crypto (equivalent):
```bash
node -e "
const { generateKeyPairSync } = require('crypto');
const fs = require('fs');
const { publicKey, privateKey } = generateKeyPairSync('ed25519');
fs.writeFileSync('founder.key.hex', privateKey.export({ type: 'pkcs8', format: 'der' }).toString('hex'), { mode: 0o600 });
fs.writeFileSync('founder.pub.hex', publicKey.export({ type: 'spki', format: 'der' }).toString('hex'));
console.log('founder.pub.hex ready to register in NEX_TRUSTED_FOUNDER_KEYS_HEX');
```
